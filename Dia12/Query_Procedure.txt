ALTER PROCEDURE ATUALIZA_SALDO ( --PARA ALTERAR UMA PROCEDURE, SÓ COLOCAR UM ALTER AO INVÉS DE CREATE.
	@IDCONTA INT,
	@VALORMOV DECIMAL(7,2),
	@TIPMOV NVARCHAR(1)
)
AS 
BEGIN
	DECLARE @SALDO DECIMAL(7,2)
	IF (EXISTS(SELECT IDCONTA FROM CONTA WHERE IDCONTA=@IDCONTA))
		IF(@TIPMOV = 'D')
			BEGIN
				INSERT INTO MOVIMENTACAO (IDCONTA, TIPMOV, VALORMOV, 
				DATAMOV) VALUES (@IDCONTA, @TIPMOV, @VALORMOV, GETDATE());

				UPDATE CONTA SET SALDO=(SELECT SALDO + @VALORMOV 
				FROM CONTA WHERE IDCONTA=@IDCONTA) WHERE IDCONTA=@IDCONTA;
			END
		ELSE IF (@TIPMOV = 'S') --PARTE DO SAQUE
			BEGIN--COMPLEXIDADE, CONSTRUINDO SOLUÇÕES
				SELECT @SALDO = SALDO FROM CONTA WHERE IDCONTA=@IDCONTA

				IF (@SALDO < @VALORMOV)
					SELECT 'SALDO INSUFICIENTE! SALDO: ' + CONVERT(NVARCHAR, @SALDO)
				ELSE
					BEGIN
						INSERT INTO MOVIMENTACAO (IDCONTA, TIPMOV, VALORMOV, 
						DATAMOV) VALUES (@IDCONTA, @TIPMOV, @VALORMOV*-1, GETDATE());

						UPDATE CONTA SET SALDO=(SELECT SALDO - @VALORMOV 
						FROM CONTA WHERE IDCONTA=@IDCONTA) WHERE IDCONTA=@IDCONTA;

					--SELECT 'SAQUEI';
						INSERT INTO MOVIMENTACAO (IDCONTA, TIPMOV, VALORMOV, 
						DATAMOV) VALUES (@IDCONTA, @TIPMOV, @VALORMOV*-1, GETDATE());--MULTIPLICAR O VALORMOV POR -1 PARA CONTABILIZAR O NEGATIVO
					END
			END
		ELSE
			SELECT 'OPÇÃO INVÁLIDA';	

	--ELSE
		--SELECT 'CONTA NÃO ENCONTRADA!'	
	
END
