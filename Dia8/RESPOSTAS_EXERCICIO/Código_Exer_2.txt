/*Soluções propostas:
1.criação da tupla AUTOR, assim a tabela livro passará a reverenciar a tabela autor, sem a necessidade da criação de 
um livro para o cadastro de um autor.
2.Criação da tupla TIPO_USUARIO, que faz com que o usuario cadastrado não precise ser rotulado com um tipo, sendo possível
cadastrar quantos tipos quiser a partir dessa solução.
3.Todas as informações do autor, incluindo seu nome, estão armazenadas na tupla AUTOR. Assim todas as vezes que algo for 
alterado lá, as tabelas que a referenciam também serão.
4.Os dados do usuário estão armazenados na tabela de usuários, sendo assim toda e qualquer atualização de dados da mesma
constará na tabela que a referenciar.
5.OBRIGATORIEDADE de ID_LIVRO na tupla EMPRESTIMO, isso faz com que após a exclusão da o ID referenciado, as colunas das
tuplas com obrigatoriedade de referência do mesmo sejam excluídas.

VANTAGENS DA NORMALIZAÇÃO:
1. Pode cadastrar quantos autores, endereços, usuários, emails e números de telefone quiser, sendo necessário
somente referenciar o cadastro através de ID, evitando escrever o mesmo dado repetidas vezes.
Telefone e emails podem estar ativos ou não, para isso usamos a tupla STATUS.
Para evitar que mais de um usuário use os mesmos email ou telefone,definirei pelo DDL como UNIQUE. */

--iniciando com o DDL
CREATE TABLE EDITORA(
ID INT NOT NULL,--SEM NENHUMA REFERENCIA AINDA.
NOME NVARCHAR(100),
PRIMARY KEY(ID)
)

CREATE TABLE STATUS(
ID INT NOT NULL,
NOME NVARCHAR(100),--SEM NENHUMA REFERENCIA AINDA.
PRIMARY KEY(ID)
)

CREATE TABLE GENERO(
ID INT NOT NULL,
NOME NVARCHAR(100),--SEM NENHUMA REFERENCIA AINDA.
PRIMARY KEY(ID)
)

CREATE TABLE AUTOR(
ID INT NOT NULL,
NOME NVARCHAR(100),--SEM NENHUMA REFERENCIA AINDA.
DATA_NASC DATE,
PRIMARY KEY (ID)
)

CREATE TABLE RUA(
ID INT NOT NULL,
NOME NVARCHAR(100),--SEM NENHUMA REFERENCIA AINDA.
PRIMARY KEY(ID)
)

CREATE TABLE BAIRRO(
ID INT NOT NULL,
NOME NVARCHAR(100),--SEM NENHUMA REFERENCIA AINDA.
PRIMARY KEY(ID)
)

CREATE TABLE CIDADE(
ID INT NOT NULL,
NOME NVARCHAR(100),--SEM NENHUMA REFERENCIA AINDA.
PRIMARY KEY(ID)
)

CREATE TABLE TIPO_USUARIO(
ID INT NOT NULL,
DESCRICAO NVARCHAR(200),--SEM NENHUMA REFERENCIA AINDA.
PRIMARY KEY(ID)
)

CREATE TABLE TELEFONE(
ID INT NOT NULL,
NUMERO NVARCHAR(18),--SEM NENHUMA REFERENCIA AINDA.
ID_STATUS INT NOT NULL,
PRIMARY KEY(ID),
UNIQUE (NUMERO)--NINGUEM PODE TER O MESMO NUMERO!
)

CREATE TABLE EMAIL(
ID INT NOT NULL,
ENDERECO NVARCHAR(100),
ID_STATUS INT NOT NULL,
UNIQUE(ENDERECO), --NINGUEM USAR O MESMO EMAIL
PRIMARY KEY(ID)
)

CREATE TABLE LIVROS(
ID INT NOT NULL,
TITULO NVARCHAR(100),
ID_AUTOR INT NOT NULL,--SEM NENHUMA REFERENCIA AINDA.
ID_EDITORA INT,
ANO_PUBLICACAO DATE,
ID_STATUS INT NOT NULL,
PRIMARY KEY(ID)
)

CREATE TABLE ENDERECO(
ID INT NOT NULL,
ID_RUA INT NOT NULL,--SEM NENHUMA REFERENCIA AINDA.
ID_BAIRRO INT NOT NULL,
ID_CIDADE INT NOT NULL,
PRIMARY KEY(ID)
)

CREATE TABLE USUARIOS(
ID INT NOT NULL,
NOME_COMPLETO NVARCHAR(100),
ID_ENDERECO INT NOT NULL,
ID_TELEFONE INT,--SEM NENHUMA REFERENCIA AINDA.
ID_EMAIL INT,
DATA_NASCIMENTO DATE NOT NULL,
ID_TIPO_USUARIO INT,
PRIMARY KEY(ID)
)

CREATE TABLE EMPRESTIMOS(
ID INT NOT NULL,
ID_USUARIO INT NOT NULL,
ID_LIVRO INT NOT NULL,--SEM NENHUMA REFERENCIA AINDA.
DATA_EMPRESTIMO DATE NOT NULL,
DATA_DEVOLUCAO DATE NOT NULL,
DATA_DEVOLUCAO_REAL DATE,
PRIMARY KEY(ID)
)

--INICIANDO CONEXOES ENTRE TABELAS

--CONEXAO LIVROS(ID_AUTOR *FK) AUTOR(ID *PK)
ALTER TABLE LIVROS
ADD CONSTRAINT FK_LIVROS_1 FOREIGN KEY (ID_AUTOR) REFERENCES AUTOR(ID);

--CONEXAO LIVROS(ID_EDITORA *FK) EDITORA(ID *PK)
ALTER TABLE LIVROS
ADD CONSTRAINT FK_LIVROS_2 FOREIGN KEY(ID_EDITORA) REFERENCES EDITORA(ID);


--CONEXOES TABELA ENDERECO
--CONEXAO ENDERECO(ID_RUA *FK) RUA(ID)
ALTER TABLE ENDERECO
ADD CONSTRAINT FK_ENDERECO_1 FOREIGN KEY(ID_RUA) REFERENCES RUA(ID)

--CONEXAO ENDERECO(ID_BAIRRO *FK) BAIRRO(ID)
ALTER TABLE ENDERECO
ADD CONSTRAINT FK_ENDERECO_2 FOREIGN KEY(ID_BAIRRO) REFERENCES BAIRRO(ID)

--CONEXAO ENDERECO(ID_CIDADE *FK) CIDADE(ID)
ALTER TABLE ENDERECO
ADD CONSTRAINT FK_ENDERECO_3 FOREIGN KEY(ID_CIDADE) REFERENCES CIDADE(ID)


--CONEXOES TABELAS EMPRESTIMOS
--CONEXAO EMPRESTIMOS(ID_USUARIO *FK) USUARIOS(ID *PK)
ALTER TABLE EMPRESTIMOS
ADD CONSTRAINT FK_EMPRESTIMOS_1 FOREIGN KEY (ID_USUARIO) REFERENCES USUARIOS(ID)

--CONEXAO EMPRESTIMOS(ID_LIVRO *FK) LIVROS(ID)
ALTER TABLE EMPRESTIMOS
ADD CONSTRAINT FK_EMPRESTIMOS_2 FOREIGN KEY (ID_LIVRO) REFERENCES LIVROS(ID)


--CONEXOES NA TABELA TELEFONE
--CONEXAO TELEFONE(ID_STATUS *FK) STATUS(ID)
ALTER TABLE TELEFONE
ADD CONSTRAINT FK_TELEFONE_1 FOREIGN KEY(ID_STATUS) REFERENCES STATUS(ID)

--CONEXAO EMAIL(ID_STATUS *FK) STATUS(ID *PK)

--CONEXAO USUARIOS(ID_ENDERECO) ENDERECO(ID)
ALTER TABLE USUARIOS
ADD CONSTRAINT FK_USUARIOS_1 FOREIGN KEY(ID_ENDERECO) REFERENCES ENDERECO(ID)

--CONEXAO USUARIOS(ID_TELEFONE *FK) TELEFONE(ID *PK)
ALTER TABLE USUARIOS
ADD CONSTRAINT FK_USUARIOS_2 FOREIGN KEY(ID_TELEFONE) REFERENCES TELEFONE(ID)

--CONEXAO USUARIOS(ID_EMAIL *FK) EMAIL(ID *PK)
ALTER TABLE USUARIOS
ADD CONSTRAINT FK_USUARIOS_3 FOREIGN KEY(ID_EMAIL) REFERENCES EMAIL(ID)

--CONEXAO USUARIOS(ID_TIPO_USUARIO *FK) TIPO_USUARIO(ID *PK)
ALTER TABLE USUARIOS
ADD CONSTRAINT FK_USUARIOS_4 FOREIGN KEY(ID_TIPO_USUARIO) REFERENCES TIPO_USUARIO(ID)
