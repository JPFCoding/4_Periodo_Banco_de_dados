 ALTER TABLE nome_da_tabela

4.Como saber se a coluna se tornou enfim uma primary 
key no SQLserver

 SELECT COLUMN_NAME
 FROM INFORMATION_SCHEMA.KEY_COLUMN_USAGE
 WHERE TABLE_NAME = 'Clientes' AND CONSTRAINT_NAME LIKE
 'PK_%';
